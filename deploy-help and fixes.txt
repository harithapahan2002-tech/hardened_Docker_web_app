----------Install Docker Compose-----------

# Remove snap docker (if u have)
sudo snap remove docker

# Install official Docker
sudo apt-get update
sudo apt-get install -y ca-certificates curl gnupg lsb-release

# Add Docker's official GPG key
sudo mkdir -p /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

# Set up repository
echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu \
  $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# Install Docker
sudo apt-get update
sudo apt-get install -y docker-ce docker-ce-cli containerd.io docker-compose-plugin

# Add your user to docker group
sudo usermod -aG docker $USER

#enable docker
sudo systemctl enable docker (if it fails just restart the system and continue to next step)

# Log out and log back in, then test
docker --version
docker compose version



-------Install Ansible----------
sudo apt install software-properties-common -y
sudo add-apt-repository --yes --update ppa:ansible/ansible
sudo apt install ansible -y
ansible --version

---------Install Trivy (Vulnerability Scanner)------
# Add Trivy repository
sudo apt-get install wget apt-transport-https gnupg lsb-release -y
wget -qO - https://aquasecurity.github.io/trivy-repo/deb/public.key | gpg --dearmor | sudo tee /usr/share/keyrings/trivy.gpg > /dev/null
echo "deb [signed-by=/usr/share/keyrings/trivy.gpg] https://aquasecurity.github.io/trivy-repo/deb $(lsb_release -sc) main" | sudo tee -a /etc/apt/sources.list.d/trivy.list

# Install Trivy
sudo apt-get update
sudo apt-get install trivy -y
trivy --version

---------------Application Not Accessible----------------------

Symptom: Browser shows "Connection refused" or "Unable to connect"
Diagnosis:
# 1. Check if nginx is running
docker compose ps nginx

# 2. Check if port is open
sudo ufw status | grep 80

# 3. Test locally on server
curl http://localhost

# 4. Check from outside
curl http://YOUR_DROPLET_IP

----Common Fixes:
# Open firewall port
sudo ufw allow 80/tcp

# Restart nginx
docker compose restart nginx

# Check nginx configuration
docker compose exec nginx nginx -t


------------Database Connection Errors----------------
Symptom: "Connection refused" or "Access denied"
Diagnosis:
# 1. Check database is running
docker compose ps database

# 2. Check database logs
docker compose logs database | grep ERROR

# 3. Test connection from webserver
docker compose exec webserver mysql -h database -u app_user 

Common Fixes:
# 1. Verify .env file exists
cat /root/suggestion-app/.env

# 2. Recreate database
docker compose down
docker volume rm suggestion-app_db_data
docker compose up -d

# 3. Check credentials match
docker compose exec database mysql -u root -p -e "SELECT user,host FROM mysql.user;"


-------------Login Not Working----------------
Symptom: Login succeeds but redirects back to login page
Solution:
# Session cookies not working (HTTP vs HTTPS mismatch)
# Edit auth.php
nano /root/suggestion-app/webserver/public/auth.php

# For HTTP: set session.cookie_secure to 0
# For HTTPS: set session.cookie_secure to 1

# Restart webserver
docker compose restart webserver

--------------High Resource Usage-----------------
Symptom: Server slow or unresponsive
Diagnosis:
# Check resource usage
docker stats

# Check system resources
htop  # or 'top'

# Check disk space
df -h

Common Fixes:
# 1. Clean up Docker
docker system prune -a

# 2. Restart containers
docker compose restart

# 3. Check logs for memory leaks
docker compose logs --tail=100
